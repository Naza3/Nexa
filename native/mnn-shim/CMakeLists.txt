cmake_minimum_required(VERSION 3.22)
project(nexa_mnn_shim LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(NEXA_MNN_SOURCE "" CACHE PATH "Exact patched private MNN checkout")
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
 message(FATAL_ERROR "Android arm64 only")
endif()
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Linux development or Android only")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../mnn-patches/identity.py --source ${NEXA_MNN_SOURCE} --verify-post RESULT_VARIABLE verified)
if(NOT verified EQUAL 0)
 message(FATAL_ERROR "Private source identity failed")
endif()
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../mnn-patches/lock.json LOCK)
string(JSON COMMIT GET "${LOCK}" upstream_commit)
string(JSON PATCH GET "${LOCK}" patch_set_sha256)
string(JSON POLICY GET "${LOCK}" policy_sha256)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/identity.h "#define NEXA_MNN_COMMIT \"${COMMIT}\"\n#define NEXA_MNN_PATCH \"${PATCH}\"\n#define NEXA_MNN_POLICY \"${POLICY}\"\n")
if("${CMAKE_C_FLAGS} ${CMAKE_CXX_FLAGS} ${CMAKE_C_FLAGS_RELEASE} ${CMAKE_CXX_FLAGS_RELEASE}" MATCHES "-[DU][ ]*NEXA_MNN_SILENT_LOGS")
 message(FATAL_ERROR "Private CPU build owns the silent logging definition; overrides are forbidden")
endif()
set(NEXA_FORBIDDEN_DIAGNOSTICS "LLM_LOG_TO_STRING|JINJA_DEBUG|DUMP_PROFILE_INFO|DEBUG_IMAGE|MNN_OPEN_TIME_TRACE|MNN_DUMP_MEMORY|EAGLE_DEBUG")
if("${CMAKE_C_FLAGS} ${CMAKE_CXX_FLAGS} ${CMAKE_C_FLAGS_RELEASE} ${CMAKE_CXX_FLAGS_RELEASE}" MATCHES "-D[ ]*(${NEXA_FORBIDDEN_DIAGNOSTICS})(=| |$)")
 message(FATAL_ERROR "Private CPU build forbids diagnostic compiler definitions")
endif()
add_compile_definitions(NEXA_MNN_SILENT_LOGS=1)
add_compile_options(-ULLM_LOG_TO_STRING -UJINJA_DEBUG -UDUMP_PROFILE_INFO)
# CPU-only, offline, text-only. FORCE prevents a reused cache enabling other backends.
foreach(flag MNN_BUILD_SHARED_LIBS MNN_SEP_BUILD MNN_OPENMP MNN_BUILD_TRAIN MNN_BUILD_DEMO
    MNN_BUILD_TOOLS MNN_BUILD_QUANTOOLS MNN_EVALUATION MNN_BUILD_CONVERTER MNN_BUILD_PROTOBUFFER
    MNN_BUILD_TEST MNN_BUILD_BENCHMARK MNN_BUILD_LLM_OMNI MNN_BUILD_OPENCV MNN_BUILD_AUDIO
    MNN_IMGCODECS MNN_BUILD_DIFFUSION MNN_INTERNAL MNN_JNI MNN_WITH_PLUGIN MNN_OPENCL
    MNN_OPENGL MNN_VULKAN MNN_METAL MNN_CUDA MNN_MUSA MNN_TENSORRT MNN_COREML MNN_NNAPI
    MNN_QNN MNN_HEXAGON MNN_ONEDNN MNN_KLEIDIAI MNN_SME2 MNN_AVX512 MNN_LLM_BUILD_DEMO
    MNN_LLM_BUILD_TEST LLM_SUPPORT_HTTP_RESOURCE BUILD_MLS MNN_USE_LOGCAT)
    set(${flag} OFF CACHE BOOL "Nexa CPU shim fixed option" FORCE)
endforeach()
set(MNN_BUILD_LLM ON CACHE BOOL "Nexa CPU shim" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "Nexa CLI shim" FORCE)
add_subdirectory("${NEXA_MNN_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mnn" EXCLUDE_FROM_ALL)
add_library(nexa-mnn-shim STATIC src/shim.cpp)
target_include_directories(nexa-mnn-shim PUBLIC include PRIVATE src ${CMAKE_CURRENT_BINARY_DIR} ${NEXA_MNN_SOURCE}/include ${NEXA_MNN_SOURCE}/transformers/llm/engine/include ${NEXA_MNN_SOURCE}/transformers/llm/engine/src ${NEXA_MNN_SOURCE}/3rd_party)
target_link_libraries(nexa-mnn-shim PUBLIC MNN)
add_executable(mnn-stream-test tests/stream_buffer_test.cpp)
target_include_directories(mnn-stream-test PRIVATE src)
add_executable(mnn-request-test tests/request_test.cpp)
target_include_directories(mnn-request-test PRIVATE ${NEXA_MNN_SOURCE}/3rd_party ${NEXA_MNN_SOURCE}/transformers/llm/engine/src ${NEXA_MNN_SOURCE}/transformers/llm/engine/include ${NEXA_MNN_SOURCE}/include)
target_link_libraries(mnn-request-test PRIVATE nexa-mnn-shim)
enable_testing()
add_test(NAME stream COMMAND mnn-stream-test)
add_test(NAME abi COMMAND mnn-request-test)
target_compile_options(mnn-stream-test PRIVATE -UNDEBUG)
target_compile_options(mnn-request-test PRIVATE -UNDEBUG)
add_executable(mnn-logging-macro-test tests/logging_macro_test.cpp)
target_include_directories(mnn-logging-macro-test PRIVATE ${NEXA_MNN_SOURCE}/include)
add_test(NAME logging-macro COMMAND mnn-logging-macro-test)
